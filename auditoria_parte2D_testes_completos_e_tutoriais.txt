def test_summary_statistics_matches_csv_table():
    """Verify metrics in summary_statistics.json match metrics_summary.csv exactly across all 20 cells."""
    json_path = os.path.join(REPO_DIR, "results", "publication_campaign", "summary_statistics.json")
    csv_path = os.path.join(REPO_DIR, "results", "publication_campaign", "tables", "metrics_summary.csv")

    assert os.path.exists(json_path), f"Missing {json_path}"
    assert os.path.exists(csv_path), f"Missing {csv_path}"

    with open(json_path, "r", encoding="utf-8") as f:
        stats_data = json.load(f)

    with open(csv_path, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        csv_rows = list(reader)

    # 5 agents x 4 workloads = 20 rows
    assert len(csv_rows) == 20, f"Expected 20 rows (5 agents x 4 workloads), got {len(csv_rows)}"

    for row in csv_rows:
        ak = row["agent"]
        w = row["workload"]
        assert ak in stats_data, f"Agent {ak} missing from summary JSON"
        assert w in stats_data[ak], f"Workload {w} missing from agent {ak} in summary JSON"
        expected = stats_data[ak][w]

        # Verify numerical consistency between CSV and JSON
        assert abs(float(row["t_task_median_ms"]) - expected["t_task_median_ms"]) < 1e-2
        assert abs(float(row["rss_peak_mb"]) - expected["rss_peak_mb"]) < 1e-2

def test_smc_power_monitor_slice_energy_synthetic():
    """Verify SMCPowerMonitor.slice_energy correctly calculates trapezoidal integration for synthetic samples."""
    from benchmark.telemetry import SMCPowerMonitor

    monitor = SMCPowerMonitor(sample_interval_ms=50)
    monitor.idle_package_mw = 1500.0
    monitor.active = True

    # 1. Test single sample fallback with elapsed_ms from timestamp interval
    t0 = 1000000000  # 1.0s in ns
    t1 = 2000000000  # 2.0s in ns (1000 ms elapsed)
    monitor._samples = [{"timestamp_ns": 1500000000, "elapsed_ms": 50.0, "package_mw": 5000.0, "cpu_mw": 3000.0, "gpu_mw": 1000.0, "ane_mw": 0.0}]

    res_single = monitor.slice_energy(t0, t1)
    assert res_single is not None
    # 5000 mW total for 1.0s = 5.0 J
    assert abs(res_single["energy_total_j"] - 5.0) < 0.1
    # 3500 mW net above 1500 mW idle for 1.0s = 3.5 J
    assert abs(res_single["energy_net_above_idle_j"] - 3.5) < 0.1

    # 2. Test multi-sample trapezoidal integration
    t_start = 0
    t_end = 300000000
    monitor._samples = [
        {"timestamp_ns": 50000000, "elapsed_ms": 50.0, "package_mw": 4000.0, "cpu_mw": 2000.0, "gpu_mw": 1000.0, "ane_mw": 0.0},
        {"timestamp_ns": 150000000, "elapsed_ms": 100.0, "package_mw": 6000.0, "cpu_mw": 3000.0, "gpu_mw": 2000.0, "ane_mw": 0.0},
        {"timestamp_ns": 250000000, "elapsed_ms": 100.0, "package_mw": 8000.0, "cpu_mw": 4000.0, "gpu_mw": 3000.0, "ane_mw": 0.0},
    ]
    res_multi = monitor.slice_energy(t_start, t_end)
    assert res_multi is not None
    assert res_multi["energy_total_j"] > 0.0
    assert res_multi["energy_net_above_idle_j"] > 0.0


def test_prompt_token_parity_across_equalized_arms(monkeypatch):
    """
    Verify prompt tokens and parity across all 4 equalized arms
    (picoclaw, openclaw_minimal, python_native, neural_baseline) across all 4 workloads (W1-W4)
    against length-sensitive mock server, and verify exact token parity in raw and warm trials files.
    """
    from benchmark.adapters import get_adapter
    from benchmark.run_unified_benchmark import start_mock_server_if_needed

    mock_port = 8092
    start_mock_server_if_needed(mock_port)
    monkeypatch.setenv("LLAMA_API_URL", f"http://127.0.0.1:{mock_port}/v1/chat/completions")

    equalized_arms = ["picoclaw", "openclaw_minimal", "python_native", "neural_baseline"]

    for w in ["W1", "W2", "W3", "W4"]:
        workload_tokens = {}
        for arm in equalized_arms:
            adapter = get_adapter(arm, REPO_DIR)
            res = adapter.run_workload(w)
            assert res["prompt_tokens"] > 0, f"{arm} {w} returned 0 prompt tokens"
            workload_tokens[arm] = res["prompt_tokens"]

        # Check that prompt tokens exist and are within reasonable parity
        for arm in equalized_arms:
            assert workload_tokens[arm] > 0

    # Verify prompt_tokens parity: strict parity in warm_cache_trials.jsonl (Regime B)
    # and pre-equalization baseline preservation (<= 5.2% variance) in raw_trials.jsonl (Regime A)
    expected_tokens_regime_b = {
        "W1": 2153,
        "W2": 2177,
        "W3": 2410,
        "W4": 22980,
    }
    expected_tokens_regime_a = {
        "W1": {2153, 2273},
        "W2": {2297, 2321},
        "W3": {2410, 2530},
        "W4": {22980, 23255, 24180},
    }

    warm_file = os.path.join(REPO_DIR, "results", "publication_campaign", "warm_cache_trials.jsonl")
    if os.path.exists(warm_file):
        with open(warm_file, "r", encoding="utf-8") as f:
            warm_records = [json.loads(line) for line in f if line.strip()]
        assert len(warm_records) == 660, f"Expected 660 records in warm_cache_trials.jsonl, got {len(warm_records)}"
        for trial in warm_records:
            assert float(trial.get("ttft_ms", 0.0)) > 0.0, f"Missing ttft_ms in warm_cache_trials.jsonl: {trial}"
            assert float(trial.get("itl_ms", 0.0)) > 0.0, f"Missing itl_ms in warm_cache_trials.jsonl: {trial}"
            w = trial.get("workload")
            a = trial.get("agent_key")
            if a in equalized_arms and w in expected_tokens_regime_b:
                assert trial.get("prompt_tokens") == expected_tokens_regime_b[w], (
                    f"warm_cache_trials.jsonl mismatch: {a} {w} has {trial.get('prompt_tokens')} != {expected_tokens_regime_b[w]}"
                )

    raw_file = os.path.join(REPO_DIR, "results", "publication_campaign", "raw_trials.jsonl")
    if os.path.exists(raw_file):
        with open(raw_file, "r", encoding="utf-8") as f:
            for line in f:
                if not line.strip():
                    continue
                trial = json.loads(line)
                w = trial.get("workload")
                a = trial.get("agent_key")
                if a in equalized_arms and w in expected_tokens_regime_a:
                    assert trial.get("prompt_tokens") in expected_tokens_regime_a[w], (
                        f"raw_trials.jsonl mismatch: {a} {w} has {trial.get('prompt_tokens')} not in {expected_tokens_regime_a[w]}"
                    )

def compute_edp(energy_joules: float, latency_seconds: float) -> float:
    """Compute Energy-Delay Product (EDP) in Joule-seconds."""
    return energy_joules * latency_seconds


def test_edp_calculation():
    """Verify Energy-Delay Product calculation (Gonzalez & Horowitz, 1996)."""
    e = 15.0  # Joules
    t = 2.0   # seconds
    edp = compute_edp(e, t)
    assert edp == 30.0


def test_cliffs_delta_and_a12_extremes():
    """Verify Cliff's delta and Vargha-Delaney A12 for strictly dominated distributions."""
    sample_a = [10.0, 11.0, 12.0]
    sample_b = [20.0, 21.0, 22.0]

    # Every element in sample_a is less than every element in sample_b
    delta_ab = cliffs_delta(sample_a, sample_b)
    assert delta_ab == -1.0
    a12_ab = vargha_delaney_a12(sample_a, sample_b)
    assert a12_ab == 0.0

    # Inverted: sample_b vs sample_a
    delta_ba = cliffs_delta(sample_b, sample_a)
    assert delta_ba == 1.0
    a12_ba = vargha_delaney_a12(sample_b, sample_a)
    assert a12_ba == 1.0


def test_cliffs_delta_and_a12_identical():
    """Verify Cliff's delta and Vargha-Delaney A12 for identical distributions."""
    sample_a = [5.0, 5.0, 5.0]
    sample_b = [5.0, 5.0, 5.0]

    delta = cliffs_delta(sample_a, sample_b)
    assert delta == 0.0
    a12 = vargha_delaney_a12(sample_a, sample_b)
    assert a12 == 0.5


def test_bootstrap_ci_coverage():
    """Verify 95% bootstrap confidence interval contains the true population mean."""
    np.random.seed(42)
    # 50 samples from N(100, 5)
    data = np.random.normal(100.0, 5.0, size=50).tolist()
    low_ci, high_ci = bootstrap_ci(data, n_resamples=2000, alpha=0.05, seed=42)

    assert low_ci < 100.0 < high_ci


def test_holm_bonferroni_adjustment():
    """Verify family-wise error rate control via Holm-Bonferroni step-down adjustment."""
    # Raw p-values: smallest gets multiplied by m, second smallest by m-1, etc.
    # Also verify enforce monotonicity: adj_p[i] <= adj_p[i+1]
    raw_p = [0.005, 0.015, 0.040, 0.200]
    adj_p = holm_bonferroni_adjust(raw_p)

    assert len(adj_p) == len(raw_p)
    # Check that adjusted p-values are >= raw p-values and <= 1.0
    for r, a in zip(raw_p, adj_p):
        assert a >= r
        assert a <= 1.0

    # Check monotonicity
    for i in range(len(adj_p) - 1):
        assert adj_p[i] <= adj_p[i + 1]


def test_mann_whitney_significance():
    """Verify two-sided Mann-Whitney U test on separated samples."""
    sample_fast = [1.2, 1.3, 1.1, 1.4, 1.25]
    sample_slow = [5.2, 5.5, 5.3, 5.8, 5.4]

    u_stat, p_val = stats.mannwhitneyu(sample_fast, sample_slow, alternative="two-sided")
    assert p_val < 0.01
    d = cliffs_delta(sample_fast, sample_slow)
    assert d == -1.0

def test_mock_files_directory_structure():
    """Verify mock_files contains exactly 50 synthetic deterministic files."""
    mock_dir = os.path.join(DATA_DIR, "mock_files")
    assert os.path.exists(mock_dir), f"Directory {mock_dir} does not exist"
    files = [f for f in os.listdir(mock_dir) if not f.startswith(".")]
    assert len(files) == 50, f"Expected 50 mock files, found {len(files)}"

        full_p = os.path.join(mock_dir, f)
        assert os.path.getsize(full_p) > 0, f"File {f} is empty"


def test_system_logs_schema_and_record_count():
    """Verify system_logs.jsonl contains 750 valid JSON lines with required fields."""
    log_file = os.path.join(DATA_DIR, "system_logs.jsonl")
    assert os.path.exists(log_file), f"Log file {log_file} does not exist"

    count = 0
    with open(log_file, "r", encoding="utf-8") as f:
        for idx, line in enumerate(f):
            count += 1
            record = json.loads(line)
            assert "timestamp" in record, f"Line {idx} missing timestamp"
            assert "subsystem" in record, f"Line {idx} missing subsystem"
            assert "level" in record, f"Line {idx} missing level"
            assert "latency_ms" in record, f"Line {idx} missing latency_ms"
            assert isinstance(record["latency_ms"], (int, float)), f"Line {idx} latency_ms not numeric"

    assert count == 750, f"Expected 750 log entries, counted {count}"


def test_rag_chunks_count_and_format():
    """Verify rag_chunks_10k.json contains valid text chunks with index keys."""
    rag_file = os.path.join(DATA_DIR, "rag_chunks_10k.json")
    assert os.path.exists(rag_file), f"RAG chunks file {rag_file} does not exist"

    with open(rag_file, "r", encoding="utf-8") as f:
        chunks = json.load(f)

    assert len(chunks) >= 100, f"Expected at least 100 chunks, found {len(chunks)}"
    assert "0" in chunks, "Index 0 missing in chunks"
    assert isinstance(chunks["0"], str) and len(chunks["0"]) > 50


def test_embedding_head_matrix_dimensions():
    """Verify embedding_head_100k.npz contains the dense projection matrix."""
    emb_file = os.path.join(DATA_DIR, "embedding_head_100k.npz")
    assert os.path.exists(emb_file), f"Embedding weights {emb_file} missing"

    data = np.load(emb_file)

def test_integrate_energy_trapezoidal():
    """Verify trapezoidal numerical integration of power samples over monotonic timestamps."""
    # 0.0s: 10 W (10000 mW)
    # 1.0s: 20 W (20000 mW) -> trapezoid area = 1.0 * (10 + 20) / 2 = 15 J
    # 2.0s: 20 W (20000 mW) -> rectangle area = 1.0 * (20 + 20) / 2 = 20 J
    # Total expected: 35.0 J
    samples = [
        {"timestamp_ns": 0, "package_mw": 10000.0, "cpu_mw": 6000.0, "gpu_mw": 4000.0, "ane_mw": 0.0},
        {"timestamp_ns": 1_000_000_000, "package_mw": 20000.0, "cpu_mw": 12000.0, "gpu_mw": 8000.0, "ane_mw": 0.0},
        {"timestamp_ns": 2_000_000_000, "package_mw": 20000.0, "cpu_mw": 12000.0, "gpu_mw": 8000.0, "ane_mw": 0.0},
    ]
    summary = integrate_energy_trapezoidal(samples)
    assert pytest.approx(summary["energy_total_j"], 0.001) == 35.0
    assert summary["duration_s"] == 2.0


def test_integrate_energy_empty_or_single():
    """Verify empty sample list yields 0.0 Joules."""

def _cmd_out(cmd: list[str]) -> str:
    try:
        return subprocess.check_output(cmd, cwd=REPO_DIR, stderr=subprocess.STDOUT, text=True).strip()
    except Exception as e:
        return f"unavailable ({e})"


def collect_environment_versions() -> dict[str, Any]:
    def find_bin(candidates: list[str]) -> str:
        for c in candidates:
            if os.path.isabs(c) and os.path.exists(c):
                return c
            elif os.path.exists(os.path.join(REPO_DIR, c)):
                return os.path.join(REPO_DIR, c)
            elif subprocess.run(["which", c], capture_output=True).returncode == 0:
                return c
        return candidates[0]

    go_bin = find_bin([os.path.join(REPO_DIR, "tools", "go", "bin", "go"), "go", "/opt/homebrew/bin/go", "/usr/local/bin/go"])
    node_bin = find_bin([os.path.join(REPO_DIR, "tools", "node", "bin", "node"), "node", "/opt/homebrew/bin/node", "/usr/local/bin/node"])
    bun_bin = find_bin([os.path.join(REPO_DIR, "tools", "bun", "bin", "bun"), os.path.expanduser("~/.bun/bin/bun"), "bun", "/opt/homebrew/bin/bun"])

    return {
        "timestamp_iso": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "git_commit": _cmd_out(["git", "rev-parse", "--short", "HEAD"]),
        "platform_system": platform.system(),

def load_records_by_cell(path: str) -> dict[tuple[str, str], list[dict[str, Any]]]:
    data: dict[tuple[str, str], list[dict[str, Any]]] = {}
    if not os.path.exists(path):
        raise RuntimeError(f"FATAL: Required trials file missing: {path}")
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            if not line.strip():
                continue
            d = json.loads(line)
            key = (d["workload"], d["agent_key"])
            data.setdefault(key, []).append(d)
    return data


def _mean(vals: list[float]) -> float:
    if not vals:
        raise RuntimeError("FATAL: Empty list in _mean")
    return sum(vals) / len(vals)


def _std(vals: list[float]) -> float:
    if len(vals) < 2:
        return 0.0
    m = _mean(vals)
    return math.sqrt(sum((x - m) ** 2 for x in vals) / (len(vals) - 1))


def _median(vals: list[float]) -> float:
    s = sorted(vals)
    n = len(s)
    if n == 0:
        raise RuntimeError("FATAL: Empty list in _median")
    if n % 2 == 1:
        return s[n // 2]
    return 0.5 * (s[n // 2 - 1] + s[n // 2])

def check_server_health(url: str = "http://127.0.0.1:8080/health") -> bool:
    req = urllib.request.Request(url)
    with urllib.request.urlopen(req, timeout=3.0) as resp:
        return resp.status == 200


def _mean(vals: list[float]) -> float:
    return sum(vals) / len(vals) if vals else 0.0


def _median(vals: list[float]) -> float:
    if not vals:
        return 0.0
    s = sorted(vals)
    n = len(s)
    return s[n // 2] if n % 2 == 1 else 0.5 * (s[n // 2 - 1] + s[n // 2])


def _std(vals: list[float]) -> float:
    if len(vals) < 2:
        return 0.0
    m = _mean(vals)
    return math.sqrt(sum((x - m) ** 2 for x in vals) / (len(vals) - 1))

        raise RuntimeError(f"FATAL: Reference campaign file missing: {path}")
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            if not line.strip():
                continue
            d = json.loads(line)
            if d.get("agent_key") in ("openclaw_full", "openclaw_minimal"):
                rec = dict(d)
                rec["regime"] = regime
                rec["t_prune_ms"] = 0.0
                records.append(rec)
    return records

def main() -> int:
    if sys.platform != "darwin":
        raise RuntimeError(f"FATAL: Physical hardware execution requires macOS Darwin, got {sys.platform}")

    workloads = ["W1", "W2", "W3", "W4"]
    tool_rag_agents = ["openclaw_tool_rag", "picoclaw_tool_rag"]

    if "--analyze-only" in sys.argv and os.path.exists(OUT_TRIALS_PATH):

        measured_tool_rag_records = []

                        t_prune_ms = float(res.get("t_prune_ms", 0.0))

                            raise RuntimeError(f"FATAL: Invalid Tool-RAG warmup telemetry: {res}")
                        t_task_ms = float(res.get("t_task_ms", 0.0))
                        t_neural_ms = float(res.get("t_neural_ms", 0.0))
                        t_orch_ms = float(res.get("t_orch_ms", max(0.0, t_task_ms - t_neural_ms)))
                        phi_orch = (t_orch_ms / t_task_ms * 100.0) if t_task_ms > 0 else 0.0
                        cached_t = int(res.get("cached_tokens", 0))
                        cache_hit = (cached_t / pt * 100.0) if pt > 0 else 0.0

                            "agent_key": ak,

                            "t_neural_ms": round(t_neural_ms, 2),
                            "t_orch_ms": round(t_orch_ms, 3),
                            "phi_orch_pct": round(phi_orch, 3),
                            "ttft_ms": round(ttft_ms, 2),
                            "itl_ms": round(itl_ms, 2),
                            "t_dispatch_ms": round(float(res.get("t_dispatch_ms", 0.0)), 3),
                            "t_prune_ms": round(t_prune_ms, 4),

                ]

                    raise RuntimeError(f"FATAL: Missing evaluated records for {reg_key} {w} {ak}")
                tasks = [float(r["t_task_ms"]) for r in sub]
                orchs = [float(r["t_orch_ms"]) for r in sub]
                prunes = [float(r.get("t_prune_ms", 0.0)) for r in sub]
                pts = [float(r["prompt_tokens"]) for r in sub]
                enets = [float(r["energy_net_above_idle_j"]) for r in sub if r.get("energy_net_above_idle_j") is not None]
                edps = [float(r.get("edp_js") or r.get("edp_net_js") or (float(r["energy_net_above_idle_j"]) * float(r["t_task_ms"]) / 1000.0)) for r in sub if r.get("energy_net_above_idle_j") is not None]
                phys = [float(r.get("proc_peak_phys_mb", 0.0)) for r in sub]

                summary[reg_key][w][ak] = {
                    "n_trials": len(sub),
                    "prompt_tokens_mean": round(_mean(pts), 1),
                    "t_prune_mean_ms": round(_mean(prunes), 4),
                    "t_task_mean_ms": round(_mean(tasks), 2),
                    "t_task_median_ms": round(_median(tasks), 2),
                    "t_task_std_ms": round(_std(tasks), 2),
                    "t_orch_mean_ms": round(_mean(orchs), 3),
                    "energy_net_mean_j": round(_mean(enets), 2),
                    "edp_mean_js": round(_mean(edps), 2),
                    "proc_peak_phys_mb_mean": round(_mean(phys), 2),
                }

            # Compute relative reductions vs openclaw_full
            full_s = summary[reg_key][w]["openclaw_full"]
            min_s = summary[reg_key][w]["openclaw_minimal"]
            for rag_ak in ("openclaw_tool_rag", "picoclaw_tool_rag"):
                rag_s = summary[reg_key][w][rag_ak]
                rag_s["latency_reduction_vs_full_54_pct"] = round(
                    (1.0 - rag_s["t_task_median_ms"] / full_s["t_task_median_ms"]) * 100.0, 2
                )
                rag_s["energy_reduction_vs_full_54_pct"] = round(
                    (1.0 - rag_s["energy_net_mean_j"] / full_s["energy_net_mean_j"]) * 100.0, 2
                )
                rag_s["edp_reduction_vs_full_54_pct"] = round(
                    (1.0 - rag_s["edp_mean_js"] / full_s["edp_mean_js"]) * 100.0, 2
                )
                rag_s["latency_reduction_vs_minimal_15_pct"] = round(
                    (1.0 - rag_s["t_task_median_ms"] / min_s["t_task_median_ms"]) * 100.0, 2
                )

    with open(OUT_SUMMARY_PATH, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2)
    print(f"✅ Saved Tool-RAG summary JSON: {OUT_SUMMARY_PATH}")

    generate_latex_table(summary)
    return 0

def compute_sha256(filepath: str) -> str:
    """Compute SHA-256 hex digest of a file in 64 KB streaming chunks."""
    h = hashlib.sha256()
    with open(filepath, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()

            or "poisson" in rel_path.lower()
            or "fig7" in rel_path
            or "fig8" in rel_path
        ):
            continue
        abs_path = os.path.join(REPO_DIR, rel_path)
        if os.path.exists(abs_path):
            updated[rel_path] = compute_sha256(abs_path)
        else:
            print(f"⚠️  [SKIP] File missing on disk, omitting from updated manifest: {rel_path}")

    with open(MANIFEST_PATH, "w", encoding="utf-8") as f:
        json.dump(updated, f, indent=2)
        f.write("\n")

    print(f"✅ [SUCCESS] Updated {len(updated)} tracked file signatures in {MANIFEST_PATH}")
    return 0


def verify_manifest() -> int:
    if "--update" in sys.argv:
        return update_manifest()

    if not os.path.exists(MANIFEST_PATH):
        print(f"❌ [ERROR] Manifest file not found: {MANIFEST_PATH}")
        return 1

    with open(MANIFEST_PATH, "r", encoding="utf-8") as f:
        manifest = json.load(f)

    total_files = len(manifest)
    print("========================================================================")
    print(" 🔐 [PT-BR] Verificação de Integridade Criptográfica (SHA-256)")
    print(f"    [EN-US] Cryptographic Artifact Integrity Verification ({total_files} files)")
    print("========================================================================")

    mismatches = []
    missing = []
    verified_count = 0

    for rel_path, exp_val in manifest.items():
        abs_path = os.path.join(REPO_DIR, rel_path)
        if not os.path.exists(abs_path):
            missing.append(rel_path)
            continue

        if isinstance(exp_val, dict):
            expected_hash = exp_val.get("sha256", "")
        else:
            expected_hash = str(exp_val)

        actual_hash = compute_sha256(abs_path)
        if actual_hash.lower() != expected_hash.lower():
            mismatches.append((rel_path, expected_hash, actual_hash))
        else:
            verified_count += 1

    if missing:
        print(f"\n❌ [ERROR] Missing files ({len(missing)}):")
        for m in missing:
            print(f"   - {m}")

    if mismatches:
        print(f"\n❌ [ERROR] Hash mismatches ({len(mismatches)}):")
        for p, exp, act in mismatches:
            print(f"   - {p}")
            print(f"     Expected: {exp}")
            print(f"     Actual:   {act}")

    if not missing and not mismatches:
        print(f"✅ [SUCCESS] All {verified_count}/{total_files} tracked files match their cryptographic SHA-256 signature!")
        print("   [PT-BR] Todos os arquivos rastreados conferem com suas assinaturas SHA-256 registradas.")
        print("========================================================================\n")
        return 0
    else:
        print(f"\n⚠️  [FAILURE] Verification failed: {verified_count}/{total_files} valid, {len(mismatches)} altered, {len(missing)} missing.")
        print("========================================================================\n")
        return 1


if __name__ == "__main__":
    sys.exit(verify_manifest())

### 1. A Analogia Central: O Carro e os Dois Motoristas
